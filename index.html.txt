<script>
// Carbon factors
const CARBON = {
    furniture: { diversion: 0.5, avoided: 1.5 },
    clothes: { diversion: 0.3, avoided: 3.5 },
    books: { diversion: 0.2, avoided: 0.8 },
    bric: { diversion: 0.4, avoided: 1.0 },
    electrical: { diversion: 0.5, avoided: 4.5 },
    mattress: { diversion: 0.6, avoided: 2.0 },
    bedframe: { diversion: 0.5, avoided: 1.5 }
};

let currentMode = 'in';
let transactions = JSON.parse(localStorage.getItem('wasteLedger') || '[]');

// Init
document.getElementById('entry-date').valueAsDate = new Date();
updateDashboard();
renderHistory();

function setMode(mode) {
    currentMode = mode;
    document.querySelectorAll('.mode-btn')[0].className = 'mode-btn' + (mode === 'in' ? ' active-in' : '');
    document.querySelectorAll('.mode-btn')[1].className = 'mode-btn' + (mode === 'out' ? ' active-out' : '');
    
    document.getElementById('dash-in').classList.toggle('hidden', mode !== 'in');
    document.getElementById('dash-out').classList.toggle('hidden', mode !== 'out');
    document.getElementById('route-group').classList.toggle('hidden', mode !== 'out');
    
    const btn = document.getElementById('save-btn');
    btn.textContent = mode === 'in' ? 'Save IN' : 'Save OUT';
    btn.className = 'submit-btn ' + (mode === 'in' ? 'submit-in' : 'submit-out');
    
    // Clear route selection
    if (mode === 'in') {
        setRoute('');
    }
}

function updateSubcats() {
    const main = document.getElementById('main-cat').value;
    const furn = document.getElementById('furniture-picker');
    const weight = document.getElementById('weight');
    
    if (main === 'furniture') {
        furn.classList.remove('hidden');
        weight.readOnly = true;
        weight.placeholder = "Auto-filled";
    } else {
        furn.classList.add('hidden');
        document.getElementById('furniture-type').value = '';
        weight.readOnly = false;
        weight.placeholder = "Enter weight (kg)";
        weight.value = '';
    }
}

function autoWeight() {
    const val = document.getElementById('furniture-type').value;
    if (!val) return;
    const parts = val.split('|');
    document.getElementById('weight').value = parts[2];
}

function setRoute(route) {
    document.getElementById('route-value').value = route;
    document.querySelectorAll('.route-btn').forEach((btn, idx) => {
        btn.className = 'route-btn';
        const routes = ['sold','recycled','binned'];
        if (route === routes[idx]) {
            btn.classList.add('selected-' + route);
        }
    });
}

function saveEntry() {
    const date = document.getElementById('entry-date').value;
    const mainCat = document.getElementById('main-cat').value;
    const furnVal = document.getElementById('furniture-type').value;
    const weight = parseFloat(document.getElementById('weight').value);
    const desc = document.getElementById('description').value;
    const route = document.getElementById('route-value').value;
    
    if (!date || !mainCat || !weight || weight <= 0) {
        alert('Please fill in date, category, and weight');
        return;
    }
    if (currentMode === 'out' && !route) {
        alert('Please select Sold, Recycled, or Binned');
        return;
    }
    
    let subCat = '', type = '';
    if (mainCat === 'furniture' && furnVal) {
        const parts = furnVal.split('|');
        subCat = parts[0];
        type = parts[1];
    } else {
        subCat = mainCat;
    }
    
    // Determine carbon factor category
    let carbonKey = mainCat;
    if (subCat === 'Mattress' || subCat === 'Bedframe') carbonKey = subCat.toLowerCase();
    
    const entry = {
        id: Date.now(),
        date,
        direction: currentMode,
        mainCat,
        subCat,
        type,
        weight,
        route: currentMode === 'out' ? route : null,
        desc,
        carbonKey
    };
    
    transactions.unshift(entry);
    localStorage.setItem('wasteLedger', JSON.stringify(transactions));
    
    // Reset form partially
    document.getElementById('furniture-type').value = '';
    document.getElementById('weight').value = '';
    document.getElementById('description').value = '';
    setRoute('');
    
    updateDashboard();
    renderHistory();
}

function updateDashboard() {
    const today = new Date().toISOString().split('T')[0];
    
    const inToday = transactions.filter(t => t.direction === 'in' && t.date === today);
    const outToday = transactions.filter(t => t.direction === 'out' && t.date === today);
    
    document.getElementById('in-today-count').textContent = inToday.length;
    document.getElementById('in-today-kg').textContent = inToday.reduce((a,b) => a + b.weight, 0).toFixed(1);
    
    document.getElementById('out-today-count').textContent = outToday.length;
    document.getElementById('out-today-kg').textContent = outToday.reduce((a,b) => a + b.weight, 0).toFixed(1);
    document.getElementById('sold-today').textContent = outToday.filter(t => t.route === 'sold').reduce((a,b) => a + b.weight, 0).toFixed(1);
    document.getElementById('recycled-today').textContent = outToday.filter(t => t.route === 'recycled').reduce((a,b) => a + b.weight, 0).toFixed(1);
    document.getElementById('binned-today').textContent = outToday.filter(t => t.route === 'binned').reduce((a,b) => a + b.weight, 0).toFixed(1);
    
    // All time carbon
    const totalIn = transactions.filter(t => t.direction === 'in').reduce((a,b) => a + b.weight, 0);
    const totalBinned = transactions.filter(t => t.direction === 'out' && t.route === 'binned').reduce((a,b) => a + b.weight, 0);
    const totalSold = transactions.filter(t => t.direction === 'out' && t.route === 'sold').reduce((a,b) => a + b.weight, 0);
    
    let diversion = 0;
    let avoided = 0;
    
    // Calculate properly by item
    transactions.filter(t => t.direction === 'in').forEach(t => {
        const factor = CARBON[t.carbonKey]?.diversion || 0.5;
        diversion += t.weight * factor;
    });
    transactions.filter(t => t.direction === 'out' && t.route === 'binned').forEach(t => {
        const factor = CARBON[t.carbonKey]?.diversion || 0.5;
        diversion -= t.weight * factor;
    });
    transactions.filter(t => t.direction === 'out' && t.route === 'sold').forEach(t => {
        const factor = CARBON[t.carbonKey]?.avoided || 1.0;
        avoided += t.weight * factor;
    });
    
    document.getElementById('diversion-total').textContent = Math.max(0, diversion).toFixed(1) + ' kg CO₂e';
    document.getElementById('avoided-total').textContent = Math.max(0, avoided).toFixed(1) + ' kg CO₂e';
}

function renderHistory() {
    const list = document.getElementById('history-list');
    if (transactions.length === 0) {
        list.innerHTML = '<p style="color:#999; text-align:center; padding:1rem;">No entries yet</p>';
        return;
    }
    
    list.innerHTML = transactions.slice(0, 50).map(t => {
        let badge = '';
        let label = '';
        if (t.direction === 'in') {
            badge = 'badge-in';
            label = 'IN';
        } else {
            badge = 'badge-' + t.route;
            label = t.route;
        }
        const typeStr = t.type ? ` (${t.type})` : '';
        return `
            <div class="history-item">
                <div>
                    <div style="font-weight:600;">${t.subCat}${typeStr}</div>
                    <div class="history-meta">${t.date} ${t.desc ? '· ' + t.desc : ''}</div>
                </div>
                <div style="text-align:right;">
                    <span class="badge ${badge}">${label}</span>
                    <div class="history-weight">${t.weight}kg</div>
                </div>
            </div>
        `;
    }).join('');
}

function exportCSV() {
    if (transactions.length === 0) {
        alert('No data to export');
        return;
    }
    const headers = ['Date','Direction','Category','Type','Weight(kg)','Route','Description'];
    const rows = transactions.map(t => [
        t.date, t.direction, t.subCat, t.type || '', t.weight, t.route || '', t.desc || ''
    ]);
    const csv = [headers, ...rows].map(r => r.map(x => `"${String(x).replace(/"/g,'""')}"`).join(',')).join('\n');
    
    const blob = new Blob([csv], {type: 'text/csv'});
    const url = URL.createObjectURL(blob);
    const a = document.createElement('a');
    a.href = url;
    a.download = `waste-ledger-${new Date().toISOString().split('T')[0]}.csv`;
    a.click();
    URL.revokeObjectURL(url);
}
