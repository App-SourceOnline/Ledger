function showPage(page) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    document.getElementById('page-' + page).classList.add('active');
    document.querySelectorAll('.nav-btn').forEach((btn, idx) => {
        btn.classList.toggle('active', (page === 'ledger' && idx === 0) || (page === 'reports' && idx === 1));
    });
    if (page === 'reports') generateReports();
}

function dismissReminder() {
    const today = new Date();
    localStorage.setItem('ledgerReminderMonth', `${today.getFullYear()}-${today.getMonth()}`);
    document.getElementById('monthly-banner').classList.add('hidden');
}

function checkMonthlyReminder() {
    const today = new Date();
    if (today.getDate() <= 3) { // Show on 1st, 2nd, 3rd
        const lastSeen = localStorage.getItem('ledgerReminderMonth');
        const currentMonth = `${today.getFullYear()}-${today.getMonth()}`;
        if (lastSeen !== currentMonth) {
            document.getElementById('monthly-banner').classList.remove('hidden');
        }
    }
}

function setMode(mode) {
    currentMode = mode;
    document.querySelectorAll('.mode-btn')[0].className = 'mode-btn' + (mode === 'in' ? ' active-in' : '');
    document.querySelectorAll('.mode-btn')[1].className = 'mode-btn' + (mode === 'out' ? ' active-out' : '');
    
    document.getElementById('dash-in').classList.toggle('hidden', mode !== 'in');
    document.getElementById('dash-out').classList.toggle('hidden', mode !== 'out');
    document.getElementById('route-group').classList.toggle('hidden', mode !== 'out');
    
    const btn = document.getElementById('save-btn');
    btn.textContent = mode === 'in' ? 'Save IN' : 'Save OUT';
    btn.className = 'submit-btn ' + (mode === 'in' ? 'submit-in' : 'submit-out');
    
    if (mode === 'in') setRoute('');
}

function updateSubcats() {
    const main = document.getElementById('main-cat').value;
    const furn = document.getElementById('furniture-picker');
    const weight = document.getElementById('weight');
    const qtyGroup = document.getElementById('quantity-group');
    
    if (main === 'furniture') {
        furn.classList.remove('hidden');
        weight.readOnly = true;
        weight.placeholder = "Auto-filled";
        qtyGroup.classList.remove('hidden');
    } else {
        furn.classList.add('hidden');
        document.getElementById('furniture-type').value = '';
        weight.readOnly = false;
        weight.placeholder = "Enter weight (kg)";
        weight.value = '';
        qtyGroup.classList.add('hidden');
        setQty(1);
    }
}

function autoWeight() {
    const val = document.getElementById('furniture-type').value;
    if (!val) return;
    const parts = val.split('|');
    document.getElementById('weight').value = parts[2];
    if (parts[0] === 'Dining Chair') {
        setQty(4);
    } else {
        setQty(1);
    }
}

function setQty(n) {
    document.getElementById('quantity').value = n;
    document.querySelectorAll('.qty-btn').forEach(btn => {
        btn.classList.toggle('active', parseInt(btn.textContent) === n);
    });
}

function setRoute(route) {
    document.getElementById('route-value').value = route;
    document.querySelectorAll('.route-btn').forEach((btn, idx) => {
        btn.className = 'route-btn';
        const routes = ['sold','recycled','binned'];
        if (route === routes[idx]) {
            btn.classList.add('selected-' + route);
        }
    });
}

function saveEntry() {
    const date = document.getElementById('entry-date').value;
    const mainCat = document.getElementById('main-cat').value;
    const furnVal = document.getElementById('furniture-type').value;
    const weight = parseFloat(document.getElementById('weight').value);
    const quantity = parseInt(document.getElementById('quantity').value) || 1;
    const desc = document.getElementById('description').value;
    const route = document.getElementById('route-value').value;
    
    if (!date || !mainCat || !weight || weight <= 0) {
        alert('Please fill in date, category, and weight');
        return;
    }
    if (currentMode === 'out' && !route) {
        alert('Please select Sold, Recycled, or Binned');
        return;
    }
    
    let subCat = '', type = '';
    if (mainCat === 'furniture' && furnVal) {
        const parts = furnVal.split('|');
        subCat = parts[0];
        type = parts[1];
    } else {
        subCat = mainCat;
    }
    
    let carbonKey = mainCat;
    if (subCat === 'Mattress' || subCat === 'Bedframe') carbonKey = subCat.toLowerCase();
    
    const entry = {
        id: Date.now(),
        date,
        direction: currentMode,
        mainCat,
        subCat,
        type,
        weight,
        quantity,
        route: currentMode === 'out' ? route : null,
        desc,
        carbonKey
    };
    
    transactions.unshift(entry);
    localStorage.setItem('wasteLedger', JSON.stringify(transactions));
    
    document.getElementById('furniture-type').value = '';
    document.getElementById('weight').value = '';
    document.getElementById('description').value = '';
    setRoute('');
    if (document.getElementById('main-cat').value === 'furniture') {
        setQty(1);
    }
    
    updateDashboard();
    renderHistory();
}

function updateDashboard() {
    const today = new Date().toISOString().split('T')[0];
    
    const inToday = transactions.filter(t => t.direction === 'in' && t.date === today);
    const outToday = transactions.filter(t => t.direction === 'out' && t.date === today);
    
    document.getElementById('in-today-count').textContent = inToday.reduce((a,b) => a + (b.quantity || 1), 0);
    document.getElementById('in-today-kg').textContent = inToday.reduce((a,b) => a + (b.weight * (b.quantity || 1)), 0).toFixed(1);
    
    document.getElementById('out-today-count').textContent = outToday.reduce((a,b) => a + (b.quantity || 1), 0);
    document.getElementById('out-today-kg').textContent = outToday.reduce((a,b) => a + (b.weight * (b.quantity || 1)), 0).toFixed(1);
    document.getElementById('sold-today').textContent = outToday.filter(t => t.route === 'sold').reduce((a,b) => a + (b.weight * (b.quantity || 1)), 0).toFixed(1);
    document.getElementById('recycled-today').textContent = outToday.filter(t => t.route === 'recycled').reduce((a,b) => a + (b.weight * (b.quantity || 1)), 0).toFixed(1);
    document.getElementById('binned-today').textContent = outToday.filter(t => t.route === 'binned').reduce((a,b) => a + (b.weight * (b.quantity || 1)), 0).toFixed(1);
    
    let diversion = 0;
    let avoided = 0;
    
    transactions.filter(t => t.direction === 'in').forEach(t => {
        const factor = CARBON[t.carbonKey]?.diversion || 0.5;
        diversion += t.weight * (t.quantity || 1) * factor;
    });
    transactions.filter(t => t.direction === 'out' && t.route === 'binned').forEach(t => {
        const factor = CARBON[t.carbonKey]?.diversion || 0.5;
        diversion -= t.weight * (t.quantity || 1) * factor;
    });
    transactions.filter(t => t.direction === 'out' && t.route === 'sold').forEach(t => {
        const factor = CARBON[t.carbonKey]?.avoided || 1.0;
        avoided += t.weight * (t.quantity || 1) * factor;
    });
    
    document.getElementById('diversion-total').textContent = Math.max(0, diversion).toFixed(1) + ' kg CO₂e';
    document.getElementById('avoided-total').textContent = Math.max(0, avoided).toFixed(1) + ' kg CO₂e';
    
    let todayDiv = 0, todayAvoid = 0;
    inToday.forEach(t => {
        const factor = CARBON[t.carbonKey]?.diversion || 0.5;
        todayDiv += t.weight * (t.quantity || 1) * factor;
    });
    outToday.filter(t => t.route === 'sold').forEach(t => {
        const factor = CARBON[t.carbonKey]?.avoided || 1.0;
        todayAvoid += t.weight * (t.quantity || 1) * factor;
    });
    outToday.filter(t => t.route === 'binned').forEach(t => {
        const factor = CARBON[t.carbonKey]?.diversion || 0.5;
        todayDiv -= t.weight * (t.quantity || 1) * factor;
    });
    
    document.getElementById('diversion-today').textContent = Math.max(0, todayDiv).toFixed(1) + ' kg CO₂e';
    document.getElementById('avoided-today').textContent = Math.max(0, todayAvoid).toFixed(1) + ' kg CO₂e';
}

function renderHistory() {
    const list = document.getElementById('history-list');
    if (transactions.length === 0) {
        list.innerHTML = '<p style="color:#999; text-align:center; padding:1rem;">No entries yet</p>';
        return;
    }
    
    list.innerHTML = transactions.slice(0, 50).map(t => {
        let badge = '';
        let label = '';
        if (t.direction === 'in') {
            badge = 'badge-in';
            label = 'IN';
        } else {
            badge = 'badge-' + t.route;
            label = t.route;
        }
        const typeStr = t.type ? ` (${t.type})` : '';
        const qtyStr = (t.quantity || 1) > 1 ? ` ×${t.quantity}` : '';
        return `
            <div class="history-item">
                <div>
                    <div style="font-weight:600;">${t.subCat}${typeStr}${qtyStr}</div>
                    <div class="history-meta">${t.date} ${t.desc ? '· ' + t.desc : ''}</div>
                </div>
                <div style="text-align:right;">
                    <span class="badge ${badge}">${label}</span>
                    <div class="history-weight">${(t.weight * (t.quantity || 1)).toFixed(1)}kg</div>
                </div>
            </div>
        `;
    }).join('');
}// NOVA UPGRADE: Fixed Excel export (no hidden columns)
function exportExcel() {
    if (transactions.length === 0) {
        alert('No data to export');
        return;
    }
    if (typeof XLSX === 'undefined') {
        alert('Excel library loading... please try again in a moment');
        return;
    }

    const wb = XLSX.utils.book_new();

    // Sheet 1: Transactions
    const transHeaders = ['Date','Direction','Category','Type','Quantity','Unit Weight (kg)','Total Weight (kg)','Route','Diversion CO2 (kg)','Avoided CO2 (kg)','Description'];
    const transRows = transactions.slice().reverse().map(t => {
        const qty = t.quantity || 1;
        const totalWeight = t.weight * qty;
        const factorDiv = CARBON[t.carbonKey]?.diversion || 0.5;
        const factorAvoid = CARBON[t.carbonKey]?.avoided || 1.0;
        
        let divCO2 = 0, avoidCO2 = 0;
        if (t.direction === 'in') {
            divCO2 = totalWeight * factorDiv;
        } else if (t.route === 'sold') {
            avoidCO2 = totalWeight * factorAvoid;
        } else if (t.route === 'binned') {
            divCO2 = -(totalWeight * factorDiv);
        }
        
        return [
            t.date,
            t.direction.toUpperCase(),
            t.subCat,
            t.type || '',
            qty,
            t.weight,
            totalWeight,
            t.route ? t.route.charAt(0).toUpperCase() + t.route.slice(1) : '',
            divCO2,
            avoidCO2,
            t.desc || ''
        ];
    });
    
    const ws1 = XLSX.utils.aoa_to_sheet([transHeaders, ...transRows]);
    ws1['!autofilter'] = {ref: "A1:K1"};
    ws1['!views'] = [{state: 'frozen', xSplit: 0, ySplit: 1, topLeftCell: 'A1', activeCell: 'A1'}];
    XLSX.utils.book_append_sheet(wb, ws1, "Transactions");

    // Sheet 2: Daily Summary
    const dailyMap = {};
    transactions.forEach(t => {
        if (!dailyMap[t.date]) {
            dailyMap[t.date] = { date: t.date, inItems:0, inKg:0, outItems:0, outKg:0, div:0, avoid:0 };
        }
        const qty = t.quantity || 1;
        const tw = t.weight * qty;
        const fDiv = CARBON[t.carbonKey]?.diversion || 0.5;
        const fAvoid = CARBON[t.carbonKey]?.avoided || 1.0;
        
        if (t.direction === 'in') {
            dailyMap[t.date].inItems += qty;
            dailyMap[t.date].inKg += tw;
            dailyMap[t.date].div += tw * fDiv;
        } else {
            dailyMap[t.date].outItems += qty;
            dailyMap[t.date].outKg += tw;
            if (t.route === 'sold') dailyMap[t.date].avoid += tw * fAvoid;
            if (t.route === 'binned') dailyMap[t.date].div -= tw * fDiv;
        }
    });
    
    const dailyHeaders = ['Date','Items In','Kg In','Items Out','Kg Out','Diversion CO2','Avoided CO2','Net CO2'];
    const dailyRows = Object.values(dailyMap).sort((a,b) => b.date.localeCompare(a.date)).map(d => [
        d.date, d.inItems, d.inKg, d.outItems, d.outKg, d.div, d.avoid, d.div + d.avoid
    ]);
    
    const ws2 = XLSX.utils.aoa_to_sheet([dailyHeaders, ...dailyRows]);
    ws2['!autofilter'] = {ref:"A1:H1"};
    ws2['!views'] = [{state: 'frozen', xSplit: 0, ySplit: 1, topLeftCell: 'A1', activeCell: 'A1'}];
    XLSX.utils.book_append_sheet(wb, ws2, "Daily Summary");

    // Sheet 3: Overview
    const totalInItems = transactions.filter(t=>t.direction==='in').reduce((a,b)=>a+(b.quantity||1),0);
    const totalInKg = transactions.filter(t=>t.direction==='in').reduce((a,b)=>a+(b.weight*(b.quantity||1)),0);
    const totalOutItems = transactions.filter(t=>t.direction==='out').reduce((a,b)=>a+(b.quantity||1),0);
    const totalOutKg = transactions.filter(t=>t.direction==='out').reduce((a,b)=>a+(b.weight*(b.quantity||1)),0);
    
    let allDiv = 0, allAvoid = 0;
    transactions.forEach(t => {
        const qty = t.quantity || 1;
        const tw = t.weight * qty;
        const fDiv = CARBON[t.carbonKey]?.diversion || 0.5;
        const fAvoid = CARBON[t.carbonKey]?.avoided || 1.0;
        if (t.direction === 'in') allDiv += tw * fDiv;
        else if (t.route === 'sold') allAvoid += tw * fAvoid;
        else if (t.route === 'binned') allDiv -= tw * fDiv;
    });
    allDiv = Math.max(0, allDiv);
    allAvoid = Math.max(0, allAvoid);
    
    const overviewData = [
        ['Metric','Value'],
        ['Total Entries', transactions.length],
        ['Total Items In', totalInItems],
        ['Total Kg In', totalInKg.toFixed(1)],
        ['Total Items Out', totalOutItems],
        ['Total Kg Out', totalOutKg.toFixed(1)],
        ['Landfill Diversion (All Time)', allDiv.toFixed(1) + ' kg CO₂e'],
        ['Avoided Manufacturing (All Time)', allAvoid.toFixed(1) + ' kg CO₂e'],
        ['Net CO2 Impact (All Time)', (allDiv + allAvoid).toFixed(1) + ' kg CO₂e']
    ];
    const ws3 = XLSX.utils.aoa_to_sheet(overviewData);
    XLSX.utils.book_append_sheet(wb, ws3, "Overview");

    XLSX.writeFile(wb, `Ledger-Carbon-Report-${new Date().toISOString().split('T')[0]}.xlsx`);
}

function generateReports() {
    if (transactions.length === 0) {
        document.getElementById('rpt-total-items').textContent = '0';
        document.getElementById('rpt-total-kg').textContent = '0';
        document.getElementById('rpt-diversion').textContent = '0';
        document.getElementById('rpt-avoided').textContent = '0';
        return;
    }
    
    // Summary numbers
    const totalInItems = transactions.filter(t=>t.direction==='in').reduce((a,b)=>a+(b.quantity||1),0);
    const totalInKg = transactions.filter(t=>t.direction==='in').reduce((a,b)=>a+(b.weight*(b.quantity||1)),0);
    
    let allDiv = 0, allAvoid = 0;
    transactions.forEach(t => {
        const qty = t.quantity || 1;
        const tw = t.weight * qty;
        const fDiv = CARBON[t.carbonKey]?.diversion || 0.5;
        const fAvoid = CARBON[t.carbonKey]?.avoided || 1.0;
        if (t.direction === 'in') allDiv += tw * fDiv;
        else if (t.route === 'sold') allAvoid += tw * fAvoid;
        else if (t.route === 'binned') allDiv -= tw * fDiv;
    });
    
    document.getElementById('rpt-total-items').textContent = totalInItems;
    document.getElementById('rpt-total-kg').textContent = totalInKg.toFixed(1);
    document.getElementById('rpt-diversion').textContent = Math.max(0, allDiv).toFixed(1);
    document.getElementById('rpt-avoided').textContent = Math.max(0, allAvoid).toFixed(1);
    
    // Category breakdown (IN items only)
    const catMap = {};
    transactions.filter(t=>t.direction==='in').forEach(t => {
        const label = t.subCat || 'Unknown';
        catMap[label] = (catMap[label] || 0) + (t.weight * (t.quantity || 1));
    });
    renderChart('chart-categories', 'doughnut', Object.keys(catMap), Object.values(catMap), [
        '#3498db','#2ecc71','#e74c3c','#f39c12','#9b59b6','#1abc9c','#e67e22','#95a5a6'
    ]);
    
    // Monthly trend (last 6 months)
    const monthMap = {};
    const today = new Date();
    for (let i = 5; i >= 0; i--) {
        const d = new Date(today.getFullYear(), today.getMonth() - i, 1);
        const key = `${d.getFullYear()}-${String(d.getMonth()+1).padStart(2,'0')}`;
        monthMap[key] = 0;
    }
    transactions.filter(t=>t.direction==='in').forEach(t => {
        const key = t.date.substring(0,7);
        if (monthMap.hasOwnProperty(key)) {
            monthMap[key] += t.weight * (t.quantity || 1);
        }
    });
    renderChart('chart-monthly', 'bar', Object.keys(monthMap), Object.values(monthMap), ['#3498db']);
    
    // Disposition routes (OUT items)
    const routes = {sold:0, recycled:0, binned:0};
    transactions.filter(t=>t.direction==='out').forEach(t => {
        if (routes.hasOwnProperty(t.route)) {
            routes[t.route] += t.weight * (t.quantity || 1);
        }
    });
    renderChart('chart-routes', 'pie', ['Sold','Recycled','Binned'], [routes.sold, routes.recycled, routes.binned], [
        '#27ae60','#f39c12','#95a5a6'
    ]);
}

function renderChart(canvasId, type, labels, data, colors) {
    const ctx = document.getElementById(canvasId).getContext('2d');
    if (charts[canvasId]) {
        charts[canvasId].destroy();
    }
    charts[canvasId] = new Chart(ctx, {
        type: type,
        data: {
            labels: labels,
            datasets: [{
                data: data,
                backgroundColor: colors,
                borderWidth: 2,
                borderColor: '#fff'
            }]
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                legend: { position: 'bottom', labels: { padding: 15, font: { size: 12 } } }
            }
        }
    });
}
